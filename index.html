class SmartHomeAgent:
    
    def __init__(self):
        self.light = False
        self.fan = False
        self.alarm = False

    def perceive(self, light_level, temperature, motion, owner_home):

        # Light Control
        if light_level < 40:
            self.light = True
        else:
            self.light = False

        # Fan Control
        if temperature > 30:
            self.fan = True
        else:
            self.fan = False

        # Security
        if motion and not owner_home:
            self.alarm = True
        else:
            self.alarm = False

    def display(self):
        print("\n------ Smart Home Status ------")
        print("Lights :", "ON" if self.light else "OFF")
        print("Fan    :", "ON" if self.fan else "OFF")
        print("Alarm  :", "ON" if self.alarm else "OFF")

heuristic = {
'A': 7,

import heapq

def astar(graph, start, goal, heuristic):
    open_list = []
    gscore = {start: 0}
    parent = {start: None}

    fscore = gscore[start] + heuristic[start]
    heapq.heappush(open_list, (fscore, start))

    while open_list:
        current_f, current = heapq.heappop(open_list)

        if current == goal:
            path = []
            while current is not None:
                path.append(current)
                current = parent[current]

            path.reverse()
            return path, gscore[goal]

        for neighbor, cost in graph[current]:
            tentative_g = gscore[current] + cost

            if neighbor not in gscore or tentative_g < gscore[neighbor]:
                gscore[neighbor] = tentative_g
                fscore = tentative_g + heuristic[neighbor]
                parent[neighbor] = current

                heapq.heappush(open_list, (fscore, neighbor))

    return None, float("inf")

# Forward Chaining: Data-driven reasoning
def forward_chaining(initial_facts):
    facts = set(initial_facts)
    changed = True

    print("\n--- Forward Chaining ---")
    print("Initial facts:", sorted(facts))

    while changed:
        changed = False

        for premises, conclusion in rules:
            if all(p in facts for p in premises):
                if conclusion not in facts:
                    facts.add(conclusion)
                    changed = True

                    rule_text = " AND ".join(premises)
                    print(
                        f"Rule fired: {rule_text} "
                        f"-> {conclusion}"
                    )

    return facts


# Backward Chaining: Goal-driven reasoning
def backward_chaining(goal, facts, visited=None):
    if visited is None:
        visited = set()

    # Goal is already established
    if goal in facts:
        return True

    # Avoid repeatedly exploring the same goal
    if goal in visited:
        return False

    visited = visited | {goal}

    # Find a rule that can establish the goal
    for premises, conclusion in rules:
        if conclusion == goal:
            # Every premise must be proved
            if all(
                backward_chaining(p, facts, visited.copy())
                for p in premises
            ):
                return True

    return False
